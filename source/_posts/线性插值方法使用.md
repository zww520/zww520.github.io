### 线性插值说明

用两点确定一条直线，然后用这条直线上的点来近似代替未知的函数值。

- **特点**：计算量极小，曲线不光滑（连接处有折角）。
- **适用场景**：实时性要求高的系统（如嵌入式仪表）、数据粗略补缺。

### 线性插值原理

线性插值（Linear Interpolation, lerp）的核心思想是：在已知的两个数据点之间，用一条直线来估算中间任意位置的值

在两个点 $(x1,y1)$ 和 $(x2,y2)$之间，求取任意 $x(x1<=x<=x2)$ 对应的值

数学公式表达
$$
y = y₀ + \frac{y₁ - y₀}{x₁ - x₀} \cdot (x - x₀)
$$

令$ t = \frac{x - x_0}{x_1 - x_0}$，可简写为
$$
y = (1 - t) \cdot y_0 + t \cdot y_1
$$

### 代码实现

#### 单点插值

给定2个点，求另一个点对应的值

```cpp
double linear_interpolation(double x, double x0, double x1, double y0, double y1) {
    if(x0 == x1) {
        return y0;
    }
	return (y0 + (x - x0) * (y1 - y0) / (x1 - x0));
}
```

#### 均匀分布数组插值

给定等间隔分布的数组，求取插值后的数组

```cpp
vector<double> linear_interpolation(const vector<double> &y, uint32_t yout_len) { 
    if(y.empty() || yout_len <= 0) {
        return vector<double>();
    }
    if(y.size() == yout_len) {
        return y;
    }
    int n = y.size();
    int m = yout_len;
    vector<double> yout(yout_len);
    double scale = (double)(n - 1) / (m - 1);
    for (int i = 0; i < m; i++) {
        // 目标点 i 对应原始数据中的浮点位置
        double pos = i * scale;
        
        // 找到左端点索引
        int idx = (int)std::floor(pos);
        double t = pos - idx;  // 权重，范围 [0, 1)
        
        // 边界处理
        if (idx >= n - 1) {
            yout[i] = y[n - 1];
        } else {
            // 线性插值公式：y = (1-t) * y0 + t * y1
            double y0 = y[idx];
            double y1 = y[idx + 1];
            yout[i] = (1.0 - t) * y0 + t * y1;
        }
    }
    return yout;
}
```

#### 不均匀分布数组插值

给定不等间隔分布的数组，求取插值后的数组

```cpp
vector<point> linear_interpolation(const vector<point> &input, uint32_t out_len) {
    auto findIndex = [input](double x) -> int {
        auto iter = std::lower_bound(input.begin(), input.end(), x, [](const point &p, double x){
            return p.x < x;
        });
        if(iter == input.end()) {
            return input.size() - 1;
        }
        return iter - input.begin();
    };

    if(input.empty() || out_len <= 0) {
        return vector<point>();
    }
    if(input.size() == out_len) {
        return input;
    }
    int n = input.size();
    int m = out_len;
    vector<point> yout(out_len);
    double diff = input[n - 1].x - input[0].x;
    double scale = (double)(diff) / (m - 1);
    for (int i = 0; i < m; i++) {
        // 目标点 i 对应原始数据中的浮点位置
        double pos = i * scale + input[0].x;
        
        // 找到左端点索引
        int idx = findIndex(pos)-1;
        if(idx < 0) {
            idx = 0;
        }
        double t = (pos - input[idx].x) / (input[idx+1].x-input[idx].x);  // 权重，范围 [0, 1)
        
        // 边界处理
        if (idx >= n - 1) {
            yout[i] = input[n - 1];
        } else {
            // 线性插值公式：y = (1-t) * y0 + t * y1
            double y0 = input[idx].y;
            double y1 = input[idx + 1].y;
            double y = (1.0 - t) * y0 + t * y1;
            std::cout<<y0<<" "<<y1<<" "<<y<<" "<<t<<" "<<idx<<" "<<pos<<std::endl;
            yout[i] = point{pos, y};
        }
    }
    return yout;
}
```
